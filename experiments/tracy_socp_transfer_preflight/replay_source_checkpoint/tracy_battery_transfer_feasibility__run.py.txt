THREAD_KEYS = ('OMP_NUM_THREADS', 'OPENBLAS_NUM_THREADS', 'MKL_NUM_THREADS',
               'VECLIB_MAXIMUM_THREADS')
if __name__ == '__main__':
    for key in THREAD_KEYS:
        os.environ[key] = '1'

import cvxpy as cp
from experiments.case118_tracy_2021 import e3_prepared
from experiments.numerical_preparation import run_qualification as q

from . import model as m

MODULE = 'experiments.tracy_battery_transfer_feasibility.run'
OUTPUT = m.HERE / 'results/feasibility_001'
OLD_ROOTS = (transfer.HERE/'results/transfer_001',
             old.m.HERE/'results/starts_001', old.m.HERE/'results/starts_002')
ARTIFACTS = ('request.json', 'raw-start.json', 'start.json', 'x0.json.gz',
             'native.json.gz', 'preparation.json.gz', 'result.json.gz')


def context():
    value = q.context()
    value['diagnostic_snapshot'] = {str(p.relative_to(transfer.ROOT)):q.digest(p)
        for p in sorted(m.HERE.iterdir()) if p.suffix in ('.py', '.md') and p.name != 'REPORT.md'}
    return value


def historical_files():
    return {str(path.relative_to(transfer.ROOT)):q.digest(path)
            for root in OLD_ROOTS for path in sorted(root.rglob('*')) if path.is_file()}


def validate_context(binding):
    if context() != binding['context']:
        raise ValueError('bound diagnostic source/environment changed')
    if historical_files() != binding['historical_files']:
        raise ValueError('historical transfer/recovery artifacts changed')


def freeze():
    current = context()
    previous = q.read(OLD_ROOTS[-1]/'binding.json')['context']
    if (current['commit'] != previous['commit'] or not previous['clean']
            or current['packages'] != previous['packages']
            or current['installed'] != previous['installed']
            or current['ipopt_library_version'] != previous['ipopt_library_version']
            or any(q.digest(transfer.ROOT/path) != digest
                   for path,digest in previous['sources'].items())
            or any(current['thread_environment'][key] != '1' for key in THREAD_KEYS)):
        raise ValueError('production source/numerical environment differs from recovery')
    if q.read(OLD_ROOTS[-1]/'invocation-finish.json')['outcome'] != 'ladder_complete':
        raise ValueError('recovery must be finalized before this diagnostic')
    data = transfer.source_inventory()
    old.m.origin_hashes(data)
    prepared = e3_prepared.verified_inputs()
    rows = []
    for trial in m.trials(data):
        view = m.construct(trial, data, prepared)
        captured = m.canonical(view[1])[-1]
        rows.append(dict(view[-1], canonical_start_sha256=transfer.object_hash(captured)))
    return dict(context=current, limits=m.LIMITS, rows=rows,
                historical_files=historical_files(), authorization='owner: Please proceed',
                note='Uncommitted isolated diagnostic snapshot; unchanged committed production sources')


def verify_view(binding, trial_id, data, prepared):
    trial = m.trials(data)[trial_id-1]
    view = m.construct(trial, data, prepared)
    captured = m.canonical(view[1])[-1]
    if dict(view[-1], canonical_start_sha256=transfer.object_hash(captured)) != binding['rows'][trial_id-1]:
        raise ValueError('declared input/schedule/start/handoff changed')
    return view, captured


def classify(record):
    if record['exception']:
        return 'infrastructure_failure'
    if record['native']['status'] != 0:
        return 'native_rejection'
    return 'feasible' if record['checks']['passed'] else 'audit_failure'


def worker(root, number):
    directory = root/f'call-{number:03d}'
    binding, request = q.read(root/'binding.json'), q.read(directory/'request.json')
    record = dict(iteration=number, request=request, exception=None, optimizer_calls=0,
                  binding_sha256=q.digest(root/'binding.json'))
    began = time.monotonic()
    def phase(name):
        q.atomic_json(directory/'phase.json', dict(phase=name, utc=q.utc(),
                                                elapsed_seconds=time.monotonic()-began))
    try:
        deadline = time.monotonic()+5
        while not (directory/'launch.json').exists() and time.monotonic()<deadline:
            time.sleep(.01)
        if (q.read(directory/'launch.json')['pid'] != os.getpid()
                or (directory/'supervision.json').exists() or (root/'STOP').exists()
                or (root/'invocation-finish.json').exists()):
            raise ValueError('worker lacks current launch authority')
        validate_context(binding)
        phase('construct')
        data, prepared = transfer.source_inventory(), e3_prepared.verified_inputs()
        view, captured = verify_view(binding, request['trial_id'], data, prepared)
        physical, solver, bindings, kwargs, schedule, raw, assigned, row = view
        record['row'] = row
        q.atomic_immutable_json(directory/'raw-start.json', dict(raw_start=serializable(raw)))
        q.atomic_immutable_json(directory/'start.json', dict(assigned_start=serializable(assigned)))
        q.atomic_gzip_json(directory/'x0.json.gz', captured)
        phase('solve')
        record['optimizer_calls'] = 1
        with warnings.catch_warnings(record=True) as caught:
            warnings.simplefilter('always')
            try:
                solver.solve(**e3_prepared.DEFAULT_PROTOCOL['ac_options'])
            except cp.error.SolverError as exc:
                if solver.preparation_evidence is None or solver.preparation_evidence.native.get('status') == 0:
                    raise
                record['solver_exception'] = str(exc)
            record['warnings'] = [dict(category=w.category.__name__, message=str(w.message)) for w in caught]
        record['preparation_evidence'] = evidence_record(solver.preparation_evidence)
        if record['preparation_evidence'] is None:
            raise ValueError('native/preparation evidence missing')
        record['native'] = record['preparation_evidence']['native']
        for name,key in (('native.json.gz','native'), ('preparation.json.gz','preparation_evidence')):
            q.atomic_gzip_json(directory/name, dict(iteration=number, data=record[key]))
        del view, physical, solver, bindings
        gc.collect()
        phase('independent_audit')
        fresh, check_capture = verify_view(binding, request['trial_id'], data, prepared)
        record['checks'] = m.assess(fresh, record['preparation_evidence'], check_capture)
        validate_context(binding)
    except Exception as exc:
        record['exception'] = f'{type(exc).__name__}: {exc}'
    record['classification'] = classify(record)
    phase('archive')
    q.atomic_gzip_json(directory/'result.json.gz', serializable(record))
    q.atomic_immutable_json(directory/'completion.json', dict(classification=record['classification'],
        artifacts={name:q.digest(directory/name) for name in ARTIFACTS if (directory/name).exists()}))
    phase('complete')


def next_trial(binding, attempts):
    if attempts and (attempts[-1]['classification'] in
            ('unfinished', 'audit_failure', 'infrastructure_failure', 'rss_limit',
             'interrupted', 'supervisor_failure')):
        return None
    completed = {a['trial_id'] for a in attempts}
    accepted = {binding['rows'][a['trial_id']-1]['job']['donor']
                for a in attempts if a['feasible']}
    return next((row['trial']['id'] for row in binding['rows']
        if row['trial']['id'] not in completed and row['job']['donor'] not in accepted), None)


def status(root):
    binding = q.read(root/'binding.json')
    validate_context(binding)
    if binding['limits'] != m.LIMITS or q.read(root/'protocol.json')['protocol'] != m.LIMITS:
        raise ValueError('diagnostic budget/protocol mismatch')
    directories = sorted(root.glob('call-*'))
    if len(directories)>m.LIMITS['max_launches']:
        raise ValueError('launch ceiling exceeded')
    attempts, used = [], 0.
    data, prepared = transfer.source_inventory(), e3_prepared.verified_inputs()
    for number,directory in enumerate(directories,1):
        request = q.read(directory/'request.json')
        expected = next_trial(binding, attempts)
        if (directory.name != f'call-{number:03d}' or request['call_id'] != number
                or request['trial_id'] != expected or request['protocol'] != q.reference(root/'protocol.json', root)
                or not 0<request['wall_seconds']<=min(m.LIMITS['wall_seconds'],m.LIMITS['total_worker_seconds']-used)):
            raise ValueError('attempt sequence/request/budget mismatch')
        item = dict(call_id=number, trial_id=request['trial_id'], feasible=False)
        if not (directory/'supervision.json').exists():
            if number != len(directories):
                raise ValueError('unfinished worker precedes another launch')
            attempts.append(dict(item,classification='unfinished'))
            break
        sup = q.read(directory/'supervision.json')
        used += sup['wall_seconds']
        resource = q.resource_evidence(directory,sup)
        item.update(classification=sup['classification'], worker_seconds=sup['wall_seconds'],
                    peak_sampled_rss_mib=sup['peak_sampled_rss_mib'], resources_verified=resource)
        if (directory/'completion.json').exists():
            completion = q.read(directory/'completion.json')
            if not {'request.json','result.json.gz'}<=set(completion['artifacts']):
                raise ValueError('completion lacks required archives')
            for name,digest in completion['artifacts'].items():
                if Path(name).name != name or q.digest(directory/name) != digest:
                    raise ValueError('completion artifact/hash mismatch')
            record = q.read(directory/'result.json.gz')
            if (record['iteration'] != number or record['request'] != request or record['binding_sha256'] != q.digest(root/'binding.json')
                    or classify(record) != record['classification']
                    or completion['classification'] != record['classification']):
                raise ValueError('result/manifest/request identity differs')
            if record['classification'] != 'infrastructure_failure':
                required = set(ARTIFACTS)
                if not required <= set(completion['artifacts']) or record['optimizer_calls'] != 1:
                    raise ValueError('completed numerical attempt lacks required evidence')
                if (record['native'] != record['preparation_evidence']['native']
                        or q.read(directory/'native.json.gz') != dict(iteration=number, data=record['native'])
                        or q.read(directory/'preparation.json.gz') != dict(iteration=number, data=record['preparation_evidence'])):
                    raise ValueError('native/preparation archives disagree')
                view,captured = verify_view(binding,request['trial_id'],data,prepared)
                if (record['row'] != view[-1] or q.read(directory/'x0.json.gz') != captured
                        or q.read(directory/'start.json')['assigned_start'] != serializable(view[6])
                        or q.read(directory/'raw-start.json')['raw_start'] != serializable(view[5])):
                    raise ValueError('physical start/input/archive mismatch')
                replay = m.assess(view,record['preparation_evidence'],captured)
                if replay != record['checks']:
                    raise ValueError('independent physical audit differs from archive')
                item.update(native_status=record['native']['status'],
                            hotspots=replay['hotspots'], physics_passed=replay['physics']['passed'],
                            pins_passed=replay['pins']['passed'])
            if sup['classification']=='exited':
                if sup['returncode']!=0 or not resource:
                    item['classification']='infrastructure_failure'
                else:
                    item['classification']=record['classification']
                    item['feasible']=record['classification']=='feasible' and sup['wall_seconds']<=request['wall_seconds']
                    if record['classification']=='feasible' and not item['feasible']:
                        item['classification']='wall_limit'
        elif sup['classification']=='exited':
            item['classification']='infrastructure_failure'
        attempts.append(item)
    next_id = next_trial(binding,attempts) if used<m.LIMITS['total_worker_seconds'] else None
    return dict(launches=len(directories), finalized=sum(a['classification']!='unfinished' for a in attempts),
        feasible_cases=sum(a['feasible'] for a in attempts), expected_cases=3,
        worker_seconds=used, attempts=attempts, next_trial=next_id,
        trials=[dict(trial=row['trial'], disposition=('finalized' if row['trial']['id'] in
            {a['trial_id'] for a in attempts if a['classification']!='unfinished'} else
            'unfinished' if row['trial']['id'] in {a['trial_id'] for a in attempts} else
            'not_needed_after_feasibility' if row['job']['donor'] in
            {binding['rows'][a['trial_id']-1]['job']['donor'] for a in attempts if a['feasible']}
            else 'unlaunched')) for row in binding['rows']])


def run(root):
    if root.resolve().parent != (m.HERE/'results').resolve():
        raise ValueError('output must be a new child of this diagnostic results directory')
    binding = freeze()
    telemetry = old.monitoring()
    root.mkdir(parents=True,exist_ok=False)
    q.atomic_immutable_json(root/'binding.json',binding)
    q.atomic_immutable_json(root/'protocol.json',dict(protocol=m.LIMITS,
        solver_options=e3_prepared.DEFAULT_PROTOCOL['ac_options'], protocol_sha256=q.digest(m.HERE/'PROTOCOL.md')))
    q.atomic_immutable_json(root/'invocation-start.json',dict(utc=q.utc(),telemetry=telemetry))
    outcome,exception='failure',None
    try:
        while True:
            progress=status(root)
            q.atomic_json(root/'report.json',progress)
            trial_id=progress['next_trial']
            if trial_id is None:
                outcome='diagnostic_complete' if all(a['classification'] in
                    ('feasible','native_rejection','wall_limit') for a in progress['attempts']) else 'failure'
                break
            telemetry=old.monitoring()
            number=progress['launches']+1
            directory=root/f'call-{number:03d}';directory.mkdir()
            wall=min(m.LIMITS['wall_seconds'],m.LIMITS['total_worker_seconds']-progress['worker_seconds'])
            request=dict(call_id=number,trial_id=trial_id,wall_seconds=wall,
                         protocol=q.reference(root/'protocol.json',root))
            q.atomic_immutable_json(directory/'request.json',request)
            q.atomic_immutable_json(directory/'monitoring.json',telemetry)
            command=[sys.executable,'-m',MODULE,'--worker',str(number),'--output',str(root)]
            q.supervise(command,directory,root,request)
    except BaseException as exc:
        exception=f'{type(exc).__name__}: {exc}'
        raise
    finally:
        q.atomic_immutable_json(root/'invocation-finish.json',dict(utc=q.utc(),outcome=outcome,exception=exception))
    return progress


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--output',type=Path,default=OUTPUT)
    parser.add_argument('--worker',type=int)
    parser.add_argument('--status',action='store_true')
    parser.add_argument('--preflight',action='store_true')
    parser.add_argument('--authorize-execution',action='store_true')
    args=parser.parse_args()
    if args.worker is not None:
        worker(args.output,args.worker)
    elif args.status:
        print(json.dumps(status(args.output),indent=2))
    elif args.preflight:
        value=freeze();print(json.dumps(dict(preflight='passed',trials=[r['trial'] for r in value['rows']],
            limits=m.LIMITS,source_commit=value['context']['commit'],clean=value['context']['clean']),indent=2))
    elif args.authorize_execution:
        print(json.dumps(run(args.output),indent=2))
    else:
        parser.error('choose --preflight, --status or --authorize-execution')


if __name__=='__main__':
    main()
